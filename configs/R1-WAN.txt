DEVICE: R1-WAN
MODEL: Cisco 2911

ROLE:
Represents the company's WAN edge router.


enable
configure terminal
hostname R1-WAN
no ip domain-lookup

interface GigabitEthernet0/0
 description LAN-TO-DSW
 ip address 176.16.0.2 255.255.255.252
 no shutdown
 ip nat inside

interface GigabitEthernet0/1
 description WAN-TO-CLOUD
 ip address 200.64.100.2 255.255.255.0
 no shutdown
 ip nat outside

access-list 1 permit 192.168.10.0 0.0.0.255
ip nat inside source list 1 interface GigabitEthernet0/1 overload
ip route 192.168.10.0 255.255.255.192 176.16.0.1
ip route 192.168.10.64 255.255.255.224 176.16.0.1
ip route 192.168.10.96 255.255.255.224 176.16.0.1
ip route 192.168.10.128 255.255.255.224 176.16.0.1
ip route 192.168.10.192 255.255.255.224 176.16.0.1
ip route 0.0.0.0 0.0.0.0 200.64.100.1
end
write memory
